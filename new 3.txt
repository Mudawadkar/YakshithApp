Step 1: Create Two ResourceGroups

az group create --name AbhayRGEastUS --location eastus
az group create --name AbhayRGWestUS --location westus




Step 2: Create Two Virtual Networks

Create VNet1-
Name: VNet-East
Region: (e.g., East US)
Address Space: 10.0.0.0/16
Subnet: 10.0.1.0/24

Create VNet2-
Name: VNet-West
Region: (e.g., West US)
Address Space: 10.1.0.0/16
Subnet: 10.1.1.0/24

az network vnet create \
  --resource-group AbhayRGEastUS \
  --name VNet-East \
  --address-prefix 10.0.0.0/16 \
  --subnet-name Subnet-East \
  --subnet-prefix 10.0.1.0/24 \
  --location eastus

az network vnet create \
  --resource-group AbhayRGWestUS \
  --name VNet-West \
  --address-prefix 10.1.0.0/16 \
  --subnet-name Subnet-West \
  --subnet-prefix 10.1.1.0/24 \
  --location westus

Step 4: Establish VNet Peering
Name: EastToWest-
Peering with remote VNet: VNet-West
Enable Allow virtual network access

Similarly, open VNet-West → Peerings → Add:

Name: WestToEast-
Peering with remote VNet: VNet-East
Enable Allow virtual network access

##ENV
# Resource groups & VNets
RG_EAST="AbhayRGEastUS"
RG_WEST="AbhayRGWestUS"
VNET_EAST="VNet-East"
VNET_WEST="VNet-West"

# Subscription id (auto-detect)
SUB_ID=$(az account show --query id -o tsv)

# Remote VNet resource IDs
REMOTE_VNET_WEST="/subscriptions/${SUB_ID}/resourceGroups/${RG_WEST}/providers/Microsoft.Network/virtualNetworks/${VNET_WEST}"
REMOTE_VNET_EAST="/subscriptions/${SUB_ID}/resourceGroups/${RG_EAST}/providers/Microsoft.Network/virtualNetworks/${VNET_EAST}"


#peering east-west
az network vnet peering create \
  --name EastToWest \
  --resource-group $RG_EAST \
  --vnet-name $VNET_EAST \
  --remote-vnet $REMOTE_VNET_WEST \
  --allow-vnet-access true


#peering west-east
az network vnet peering create \
  --name WestToEast \
  --resource-group $RG_WEST \
  --vnet-name $VNET_WEST \
  --remote-vnet $REMOTE_VNET_EAST \
  --allow-vnet-access true

Step 6: Create a Custom RBAC Role (Compute Operator)

json file
{
  "Name": "Compute Operator (Custom)",
  "IsCustom": true,
  "Description": "Can start/restart VMs and read network/storage info",
  "Actions": [
    "Microsoft.Compute/virtualMachines/start/action",
    "Microsoft.Compute/virtualMachines/restart/action",
    "Microsoft.Compute/virtualMachines/read",
    "Microsoft.Network/*/read",
    "Microsoft.Storage/*/read"
  ],
  "NotActions": [],
  "DataActions": [],
  "NotDataActions": [],
  "AssignableScopes": [
    "/subscriptions/<your-subscription-id>"
  ]
}


Assign the role to John Doe

ASSIGNEE="john.doe@<yourtenant>.onmicrosoft.com"
az role assignment create \
  --assignee "$ASSIGNEE" \
  --role "Compute Operator (Custom)" \
  --scope "/subscriptions/${SUB_ID}"
